template <typename rads_t>
std::vector<uint_t> Delta_core_t<rads_t>::calculate_table_of_deepest_nodes(sdsl::cst_sada<>& cst, table_t& marked)
{
    std::cout << "We are in calculate_table_of_best" <<std::endl;
    std::vector<uint_t> table_of_best(cst.csa.size());
    std::stack<std::tuple<node_type, uint_t, node_type>> stack;
    node_type root = cst.root();
    stack.push({root, 0, root});
    while(!stack.empty()){
        std::tuple state = stack.top();
        node_type v = std::get<0>(state);
        uint_t val = std::get<1>(state);
        node_type candidate = std::get<2>(state);
        stack.pop();
        uint_t id_v = cst.id(v);
        uint_t tmp = marked[id_v][2] + marked[id_v][3];
        if (tmp > val){
            candidate = v;
            val = tmp;
        }
        if (cst.is_leaf(v)){
            uint_t suff_numb = cst.sn(v);
            table_of_best[suff_numb] = cst.id(candidate);
        } else {
            for (uint8_t y = 0; y < SIGMA; y++){
                node_type w = cst.child(v, ALPHABET[y]);
                if (w != root){   
                    stack.push({w, val, candidate});
                }
            }
        }
    }

    return table_of_best;
}

template <typename rads_t>
void Delta_core_t<rads_t>::write_into_marked_1(uint_t id_v, uint_t v_depth, uint_t i, table_t& marked)
{   
    marked[id_v][0] = 1;
    marked[id_v][1] = i; // The starting position of the suffix of R concerned rn
    marked[id_v][2] = 0; // Offset on the edge = number of matched chars on it
    marked[id_v][3] = v_depth;
}

template <typename rads_t>
void Delta_core_t<rads_t>::write_into_marked_0(uint_t id_w, uint_t v_depth, uint_t i, uint_t cur_match, table_t& marked)
{
    if (marked[id_w][0] == 0 && cur_match >= marked[id_w][2]){
        marked[id_w][1] = i; // The starting position of the suffix of R concerned rn
        marked[id_w][2] = cur_match; // Offset on the edge = number of matched chars on it
        marked[id_w][3] = v_depth; // depth of the anscestor
    }
}


template <typename rads_t>
void Delta_core_t<rads_t>::extend(node_type& v, uint_t& i, table_t& marked, sdsl::cst_sada<>& suff_tree, uint_t& m)
{
    while (i < m){
        uint_t id_v = suff_tree.id(v);
        uint_t v_depth = suff_tree.depth(v);
        if (suff_tree.is_leaf(v)){
            v_depth -= 1;
        }
        write_into_marked_1(id_v, v_depth, i, marked);
        i++;
        v = suff_tree.sl(v);
    }
    return ;
}

template <typename rads_t>
Breakup_result Delta_core_t<rads_t>::breakup_delta_buffer()
{
    std::cout << "We are in breakup_delta_buffer" << std::endl;
    uint_t m = R.size();
    uint_t b = 0;
    sdsl::cst_sada<> suff_tree;
    sdsl::construct_im(suff_tree, batch, 1);
    table_t marked(suff_tree.nodes(), std::vector<uint_t> (4));
    uint_t i = 0;
    uint_t j = 0;
    node_type v = suff_tree.root();
    while (i < m){
        while (j < m){
            uint_t L = suff_tree.is_leaf(v) ? suff_tree.depth(v) - 1 : suff_tree.depth(v);
            if (j != i + L) {
                std::cout << "INVARIANT BROKEN: i=" << i << " j=" << j
                        << " id(v)=" << suff_tree.id(v)
                        << " is_leaf=" << suff_tree.is_leaf(v)
                        << " depth=" << suff_tree.depth(v)
                        << " expected j=" << (i + L) << std::endl;
            }
            if (suff_tree.is_leaf(v)){
                std::cout << "WE REACHED A LEAF" << std::endl;
                uint_t id_v = suff_tree.id(v);
                uint_t v_depth = suff_tree.depth(v) - 1;
                write_into_marked_1(id_v, v_depth, i, marked);
                j = i + v_depth;
                i++; 
                v = suff_tree.sl(v);
                break;
            }

            uint_t id_v = suff_tree.id(v);
            uint_t v_depth  = suff_tree.depth(v);
            write_into_marked_1(id_v, v_depth, i, marked);

            // Trying to read a new char
            char c = ref_at(j);
            node_type w = suff_tree.child(v, c); 
            uint_t cur_match = 0;
            int edge_length = suff_tree.depth(w) - v_depth;
            if (edge_length <= 0){
                j = std::max(i + marked[suff_tree.id(v)][3], i + 1);
                ++i;
                v = suff_tree.sl(v);
                if (j >= m) 
                    extend(v, i, marked, suff_tree, m);
                break;
            }
            while (cur_match < edge_length && j + cur_match < m && suff_tree.edge(w,v_depth + cur_match + 1) == ref_at(j + cur_match)){  //1-based indexing on edges
                ++cur_match;
            }
            if (cur_match < edge_length){
                uint_t id_w = suff_tree.id(w);
                write_into_marked_0(id_w, v_depth, i, cur_match, marked);
                j = std::max(i + marked[suff_tree.id(v)][3], i + 1);
                ++i;
                v = suff_tree.sl(v);
                break;
            } else{
                j += edge_length;
                uint_t id_w = suff_tree.id(w);
                uint_t w_depth = v_depth + edge_length;    // == suff_tree.depth(w)
                write_into_marked_1(id_w, w_depth, i, marked);
                v = w;
                if (j >= m){
                    extend(v, i, marked, suff_tree, m);
                    break;
                }
            }
        }
    }
    uint_t root_id = suff_tree.id(suff_tree.root());
    std::cout << "root_id = " << root_id << std::endl;
    for (uint8_t y = 0; y < SIGMA; y++){
        node_type child = suff_tree.child(suff_tree.root(), ALPHABET[y]);
        if (child != suff_tree.root()){
            uint_t cid = suff_tree.id(child);
            std::cout << "child '" << ALPHABET[y] << "' id=" << cid
                    << " marked={" << marked[cid][0] << "," << marked[cid][1]
                    << "," << marked[cid][2] << "," << marked[cid][3] << "}"
                    << std::endl;
        }
    }
    std::vector<uint_t> table_of_best = calculate_table_of_deepest_nodes(suff_tree, marked);
    return Breakup_result {std::move(marked), std::move(table_of_best)};
}
